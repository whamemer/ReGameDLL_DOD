# Options of the build, one per line: NAME=VALUE # description
# ON and OFF make a compile definition (ON) or nothing (OFF), any other value becomes a variable of the build.
# The fixes of the bugs of the original game are added here, each one under its own macro.

FIX_ENTITY_HASH_OVERFLOW=ON # The hash of a classname can point one element behind the table of entities, the table gets one more element
FIX_CAPTURE_AREA_DIVIDE_BY_ZERO=ON # The progress bar of a capture area divides by the number of capturing players, which is 0 when the capturing team left the area (SIGFPE in the original), the update is skipped

FIX_TALKMONSTER_NULL_CINE=ON # A talking monster in a scripted state without its scripted_sequence crashed when a player pressed USE on it (hlsdk-portable 2d0ef99), it does not follow then
FIX_BREAKABLE_NO_GIB_MODEL=ON # A func_breakable of the material None without a gib model precached an uninitialized pointer (hlsdk-portable 1866872), it has no gibs then
FIX_SPEAKER_UNKNOWN_PRESET=ON # A speaker with a preset above 12 read an uninitialized string (hlsdk-portable 52b067f), it says nothing then
FIX_ANIMATION_INVALID_SEQUENCE=ON # A negative sequence read before the sequence table of the model, SetBlending did not check the sequence at all (hlsdk-portable 63e3769)
FIX_SENTENCES_OVERFLOW=ON # The sentence 1537 of sentences.txt was written behind the table of the names (hlsdk-portable cb51d2a)
FIX_FUNC_TANK_BARREL_NAN=ON # A func_tank aiming at a target closer than the offset of its barrel took sqrt of a negative number and got a NaN angle (hlsdk-portable 1394637)
DEFAULT_SV_ROLLANGLE=ON # Xash3D starts with sv_rollangle 0 (GoldSrc with 2), so the view does not tilt in a strafe, the library sets 2 at start when the value is 0

GAMEDIR=dod # Gamedir path

# keys below must match liblist.gam/gameinfo.txt
SERVER_INSTALL_DIR=dlls # Where to put server dll
SERVER_LIBRARY_NAME=dod # Default server dll name
